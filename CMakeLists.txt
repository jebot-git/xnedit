cmake_minimum_required(VERSION 3.20)
project(xnedit_xr VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DE_XR_TOOLKIT "FLTK" CACHE STRING "UI toolkit (FLTK)")
if(NOT DE_XR_TOOLKIT STREQUAL "FLTK")
 message(FATAL_ERROR "Use the upstream makefiles explicitly to build the legacy Motif frontend")
endif()
option(XNEDIT_ENABLE_X11 "Build the optional X11 backend integration" OFF)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_package(OpenGL REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
find_package(Threads REQUIRED)
add_library(xnedit-core STATIC source/textBuf.c source/regularExp.c util/nedit_malloc.c source/editorconfig.c util/pathutils.c util/ec_glob.c)
target_compile_definitions(xnedit-core PUBLIC XNEDIT_FLTK)
target_link_libraries(xnedit-core PUBLIC PkgConfig::FONTCONFIG)
target_include_directories(xnedit-core PUBLIC source util)
add_library(xnedit-document STATIC fltk/document.cpp fltk/highlight.cpp fltk/search.cpp fltk/remote.cpp)
target_link_libraries(xnedit-document PUBLIC xnedit-core PkgConfig::GIO OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_include_directories(xnedit-document PUBLIC fltk)
add_executable(xnedit fltk/main.cpp)
target_compile_options(xnedit PRIVATE -Wall -Wextra)
target_compile_definitions(xnedit PRIVATE XNEDIT_ENABLE_X11=$<BOOL:${XNEDIT_ENABLE_X11}>)
target_link_libraries(xnedit PRIVATE xnedit-document fltk::fltk-shared fltk::images-shared fltk::gl-shared OpenGL::GL PkgConfig::GIO PkgConfig::FONTCONFIG)
include(fltk/desktop-menu.cmake)
de_xr_desktop_menu(xnedit)
set_target_properties(xnedit PROPERTIES INSTALL_RPATH "/opt/fltk-vr/toolkit/lib")
install(TARGETS xnedit RUNTIME DESTINATION bin)
install(FILES packaging/org.domed.XNEdit.desktop DESTINATION share/applications)
install(FILES LICENSE README-FLTK.md NOTICE-FLTK.md UPSTREAM.json DESTINATION share/doc/xnedit-xr)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(remote-driver tests/remote-driver.cpp)
 target_link_libraries(remote-driver PRIVATE xnedit-document)
 add_test(NAME gemini-titan-transfer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/remote.py $<TARGET_FILE:remote-driver>)
 set_tests_properties(gemini-titan-transfer PROPERTIES TIMEOUT 60)
 add_test(NAME spartan-transfer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/spartan.py $<TARGET_FILE:remote-driver>)
 set_tests_properties(spartan-transfer PROPERTIES TIMEOUT 60)
 add_executable(document-test tests/document.cpp)
 target_link_libraries(document-test PRIVATE xnedit-document)
 target_compile_options(document-test PRIVATE -UNDEBUG)
 add_test(NAME editor-document COMMAND document-test)
 add_executable(highlight-test tests/highlight.cpp)
 target_link_libraries(highlight-test PRIVATE xnedit-document)
 target_compile_options(highlight-test PRIVATE -UNDEBUG)
 add_test(NAME syntax-highlighting COMMAND highlight-test)
 set_tests_properties(syntax-highlighting PROPERTIES TIMEOUT 20)
 add_executable(search-test tests/search.cpp)
 target_link_libraries(search-test PRIVATE xnedit-document)
 target_compile_options(search-test PRIVATE -UNDEBUG)
 add_test(NAME find-replace COMMAND search-test)
 # Run by test-wayland.py inside its private compositor, not on the user display.
 add_executable(editor-viewport-test tests/editor-viewport.cpp fltk/desktop-menu.cpp
   "${CMAKE_CURRENT_BINARY_DIR}/appmenu-protocol.c" "${CMAKE_CURRENT_BINARY_DIR}/appmenu-client.h")
 target_compile_options(editor-viewport-test PRIVATE -UNDEBUG)
 target_compile_definitions(editor-viewport-test PRIVATE XNEDIT_ENABLE_X11=$<BOOL:${XNEDIT_ENABLE_X11}>)
 target_include_directories(editor-viewport-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_link_libraries(editor-viewport-test PRIVATE xnedit-document fltk::fltk-shared
   fltk::images-shared fltk::gl-shared OpenGL::GL PkgConfig::GIO PkgConfig::FONTCONFIG PkgConfig::DESKTOPMENU)
 if(XNEDIT_ENABLE_X11)
   target_link_libraries(editor-viewport-test PRIVATE PkgConfig::DESKTOPMENU_X11)
 endif()
endif()
